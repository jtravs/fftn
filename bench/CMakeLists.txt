# fbench: driver, one backend binary per library, catalogues (task 0.6, DESIGN.md 16).
# Each backend is an executable fbench-<library> made of the shared harness and one
# backend source; bench/fbench.py finds them in <build>/bench/.

# Build identity, reported by every backend's `describe`.

string(STRIP "${_fbench_flags}" _fbench_flags)
if(NOT FBENCH_SOURCE_COMMIT)
  # The driver passes the commit when it builds one from git archive (no repository there).
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
                  OUTPUT_VARIABLE FBENCH_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE
                  ERROR_QUIET RESULT_VARIABLE _fbench_git)
  if(NOT _fbench_git EQUAL 0)
    set(FBENCH_SOURCE_COMMIT "unknown")
  endif()
endif()

function(fbench_add_backend library)
  add_executable(fbench-${library} backends/${library}.cpp)
  fftn_configure_target(fbench-${library})
  target_include_directories(fbench-${library} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(fbench-${library} PRIVATE fbench_harness ${ARGN})
endfunction()

fbench_add_backend(fftn fftn::fftn)
# The FFTN backends build plans through the private plan core, so that an arm can set its
# build settings (the machine profile, chunking) for the threading experiments.

# fbench-fftn_omp: the same backend with THREADS through fftn::omp::executor (fftn/omp.hpp),
# linked with the compiler's OpenMP runtime and nothing else that brings one (one runtime per
# binary, DESIGN.md 16.1). Not built under ThreadSanitizer, whose reports on uninstrumented
# runtimes are false (cmake/FftnTestDeps.cmake).
option(FFTN_BENCH_OPENMP "Build the OpenMP fbench backend and threading harness" ON)
if(FFTN_BENCH_OPENMP AND NOT FFTN_SANITIZE MATCHES "thread")
  find_package(OpenMP COMPONENTS CXX)
endif()
# Links OpenMP into a benchmark target, with the tests' clang-tidy include fix when present.
function(fbench_link_openmp target)
  if(COMMAND fftn_link_openmp)
    fftn_link_openmp(${target})
  else()
    target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
  endif()
endfunction()
if(TARGET OpenMP::OpenMP_CXX AND NOT FFTN_SANITIZE MATCHES "thread")
  add_executable(fbench-fftn_omp backends/fftn.cpp)
  fftn_configure_target(fbench-fftn_omp)
  target_include_directories(fbench-fftn_omp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
                                                     ${PROJECT_SOURCE_DIR}/src)
  target_compile_definitions(fbench-fftn_omp PRIVATE FBENCH_FFTN_OMP)
  target_link_libraries(fbench-fftn_omp PRIVATE fbench_harness fftn::fftn ${CMAKE_DL_LIBS})
  fbench_link_openmp(fbench-fftn_omp)
  set(_fbench_fftn_omp ON)
  message(STATUS "fbench: fftn_omp backend (OpenMP ${OpenMP_CXX_VERSION})")
else()
  set(_fbench_fftn_omp OFF)
  message(STATUS "fbench: fftn_omp backend not built (no OpenMP, or a ThreadSanitizer build)")
endif()

# The harness of experiment T1 (bench/t1/): the fixed costs of team executions. It is
# not an fbench backend (smoke runs every fbench-* of the build), hence its name; the OpenMP
# variant links the compiler's runtime and nothing else that brings one.
find_package(Threads REQUIRED)
function(fbench_add_t1 name)
  add_executable(${name} t1/t1_costs.cpp)
  fftn_configure_target(${name})
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
                             ${CMAKE_CURRENT_BINARY_DIR} ${PROJECT_SOURCE_DIR}/src)
  target_link_libraries(${name} PRIVATE fftn::fftn Threads::Threads)
endfunction()
fbench_add_t1(t1_costs)
if(_fbench_fftn_omp)
  fbench_add_t1(t1_costs_omp)
  target_compile_definitions(t1_costs_omp PRIVATE T1_WITH_OPENMP)
  fbench_link_openmp(t1_costs_omp)
endif()

# FFTW only from an explicit installation: never a default search path, so that a
# handicapped build (Homebrew's FFTW has no SIMD on arm64) cannot be picked up by accident
# (AGENTS.md section 6). Set FFTN_BENCH_FFTW_ROOT as a CMake or environment variable.
set(_fbench_fftw_root "${FFTN_BENCH_FFTW_ROOT}")
if(NOT _fbench_fftw_root)
  set(_fbench_fftw_root "$ENV{FFTN_BENCH_FFTW_ROOT}")
endif()
if(_fbench_fftw_root)
  find_path(_fbench_fftw_include fftw3.h PATHS ${_fbench_fftw_root}/include NO_DEFAULT_PATH
            NO_CACHE)
  find_library(_fbench_fftw_lib fftw3 PATHS ${_fbench_fftw_root}/lib ${_fbench_fftw_root}/lib64
               NO_DEFAULT_PATH NO_CACHE)
  if(NOT _fbench_fftw_include OR NOT _fbench_fftw_lib)
    message(FATAL_ERROR "fbench: no double-precision FFTW under ${_fbench_fftw_root}")
  endif()
  fbench_add_backend(fftw ${_fbench_fftw_lib} ${CMAKE_DL_LIBS})
  target_include_directories(fbench-fftw SYSTEM PRIVATE ${_fbench_fftw_include})
  target_compile_definitions(fbench-fftw PRIVATE "FBENCH_FFTW_ROOT=\"${_fbench_fftw_root}\"")
  message(STATUS "fbench: FFTW backend against ${_fbench_fftw_lib}")
else()
  message(STATUS "fbench: FFTW backend not built (set FFTN_BENCH_FFTW_ROOT to an installation)")
endif()

# ---- Further competitors (task 0.7; DESIGN.md 16.2) ----------------------------------------
# Each is built only when configured, from a CMake or environment variable of the same name,
# and never from a default search path (AGENTS.md section 6). How each library is built is in
# bench/README.md ("Competitor backends").

# The value of a configuration variable: the CMake variable, else the environment's.
function(fbench_setting name out)
  if(DEFINED ${name} AND NOT "${${name}}" STREQUAL "")
    set(${out} "${${name}}" PARENT_SCOPE)
  else()
    set(${out} "$ENV{${name}}" PARENT_SCOPE)
  endif()
endfunction()

set(_fbench_competitors "")

# pocketfft (BSD-3-Clause), header-only: fetched pinned by SHA-256 (offline:
# FETCHCONTENT_SOURCE_DIR_FBENCH_POCKETFFT, a directory holding pocketfft_hdronly.h). Compiled
# into its backend with the build's vector extension, as a user compiling it would: on x86-64
# for the build machine (FFTN_BENCH_POCKETFFT_ARCH_FLAGS overrides). Benchmark binaries are
# run where they are built and never distributed.
fbench_setting(FFTN_BENCH_POCKETFFT _fbench_pocketfft)
if(_fbench_pocketfft)
  include(FetchContent)
  set(_fbench_pocketfft_commit 4d865cba24359b05b37da4e75aee24498c3b78e7)
  FetchContent_Declare(fbench_pocketfft
    URL https://raw.githubusercontent.com/mreineck/pocketfft/${_fbench_pocketfft_commit}/pocketfft_hdronly.h
    URL_HASH SHA256=518a95c552aeaee352aa30122137cd9e135d23c91e2863d9272aaa758744c86d
    DOWNLOAD_NO_EXTRACT TRUE
    SOURCE_SUBDIR fftn-unpack-only)
  FetchContent_MakeAvailable(fbench_pocketfft)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    set(_fbench_pocketfft_arch_default "-march=native")
  else()
    set(_fbench_pocketfft_arch_default "")
  endif()
  set(FFTN_BENCH_POCKETFFT_ARCH_FLAGS "${_fbench_pocketfft_arch_default}" CACHE STRING
    "Instruction-set flags for the pocketfft backend (benchmark binary only)")
  separate_arguments(_fbench_pocketfft_flags UNIX_COMMAND "${FFTN_BENCH_POCKETFFT_ARCH_FLAGS}")
  fbench_add_backend(pocketfft)
  target_include_directories(fbench-pocketfft SYSTEM PRIVATE ${fbench_pocketfft_SOURCE_DIR})
  # The library's code is compiled here, so it gets what a user's build would give it:
  # contraction to FMA allowed (FFTN's own -ffp-contract=off is for FFTN's code).
  target_compile_options(fbench-pocketfft PRIVATE ${_fbench_pocketfft_flags} -ffp-contract=fast)
  target_compile_definitions(fbench-pocketfft PRIVATE
    "FBENCH_POCKETFFT_COMMIT=\"${_fbench_pocketfft_commit}\""
    "FBENCH_POCKETFFT_ARCH_FLAGS=\"${FFTN_BENCH_POCKETFFT_ARCH_FLAGS}\"")
  list(APPEND _fbench_competitors pocketfft)
  message(STATUS "fbench: pocketfft backend (flags: '${FFTN_BENCH_POCKETFFT_ARCH_FLAGS}')")
else()
  message(STATUS "fbench: pocketfft backend not built (set FFTN_BENCH_POCKETFFT=ON)")
endif()

# KFR (GPL-2.0-or-later or commercial), from an installation of KFR's own CMake build.
fbench_setting(FFTN_BENCH_KFR_ROOT _fbench_kfr_root)
if(_fbench_kfr_root)
  find_package(KFR CONFIG REQUIRED PATHS ${_fbench_kfr_root} NO_DEFAULT_PATH)
  # A multi-architecture build installs one kfr_dft_<arch> library per instruction set.
  set(_fbench_kfr_archs "")
  foreach(_arch IN ITEMS sse2 sse41 avx avx2 avx512)
    if(TARGET kfr_dft_${_arch})
      list(APPEND _fbench_kfr_archs ${_arch})
    endif()
  endforeach()
  if(NOT _fbench_kfr_archs)
    set(_fbench_kfr_archs "single target")
  endif()
  string(REPLACE ";" "," _fbench_kfr_archs "${_fbench_kfr_archs}")
  fbench_add_backend(kfr kfr_dft)
  target_compile_definitions(fbench-kfr PRIVATE "FBENCH_KFR_ROOT=\"${_fbench_kfr_root}\""
                                                "FBENCH_KFR_ARCHS=\"${_fbench_kfr_archs}\"")
  list(APPEND _fbench_competitors kfr)
  message(STATUS "fbench: KFR ${KFR_VERSION} backend from ${_fbench_kfr_root} (${_fbench_kfr_archs})")
else()
  message(STATUS "fbench: KFR backend not built (set FFTN_BENCH_KFR_ROOT to an installation)")
endif()

# Intel oneMKL (DFTI), from an installation root: LP64 interface, sequential threading.
fbench_setting(FFTN_BENCH_MKL_ROOT _fbench_mkl_root)
if(_fbench_mkl_root)
  find_path(_fbench_mkl_include mkl_dfti.h PATHS ${_fbench_mkl_root}/include NO_DEFAULT_PATH
            NO_CACHE)
  set(_fbench_mkl_libs "")
  foreach(_lib IN ITEMS mkl_intel_lp64 mkl_sequential mkl_core)
    find_library(_fbench_mkl_${_lib} ${_lib} PATHS ${_fbench_mkl_root}/lib
                 ${_fbench_mkl_root}/lib/intel64 NO_DEFAULT_PATH NO_CACHE)
    if(NOT _fbench_mkl_${_lib})
      message(FATAL_ERROR "fbench: no ${_lib} under ${_fbench_mkl_root}")
    endif()
    list(APPEND _fbench_mkl_libs ${_fbench_mkl_${_lib}})
  endforeach()
  if(NOT _fbench_mkl_include)
    message(FATAL_ERROR "fbench: no mkl_dfti.h under ${_fbench_mkl_root}/include")
  endif()
  find_package(Threads REQUIRED)
  fbench_add_backend(mkl ${_fbench_mkl_libs} Threads::Threads ${CMAKE_DL_LIBS} m)
  target_include_directories(fbench-mkl SYSTEM PRIVATE ${_fbench_mkl_include})
  target_compile_definitions(fbench-mkl PRIVATE "FBENCH_MKL_ROOT=\"${_fbench_mkl_root}\"")
  list(APPEND _fbench_competitors mkl)
  message(STATUS "fbench: MKL backend from ${_fbench_mkl_root}")
else()
  message(STATUS "fbench: MKL backend not built (set FFTN_BENCH_MKL_ROOT to a oneMKL root)")
endif()

# FFTL.jl through libfftl (MIT), from a libfftl directory built with its build.sh. The
# embedded Julia runtime catches SIGSEGV for its garbage collector's safepoints, which the
# sanitizers' handlers break, so it is not built in sanitizer builds.
fbench_setting(FFTN_BENCH_FFTL_ROOT _fbench_fftl_root)
if(_fbench_fftl_root AND FFTN_SANITIZE)
  message(STATUS "fbench: FFTL backend not built in a sanitizer build")
elseif(_fbench_fftl_root)
  find_path(_fbench_fftl_include fftl.h PATHS ${_fbench_fftl_root}/include NO_DEFAULT_PATH
            NO_CACHE)
  find_library(_fbench_fftl_lib fftl PATHS ${_fbench_fftl_root}/build/lib
               ${_fbench_fftl_root}/lib NO_DEFAULT_PATH NO_CACHE)
  if(NOT _fbench_fftl_include OR NOT _fbench_fftl_lib)
    message(FATAL_ERROR "fbench: no include/fftl.h and build/lib/libfftl under "
                        "${_fbench_fftl_root}")
  endif()
  if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64")
    set(_fbench_fftl_target_default "apple-m1 (build.sh default)")
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    set(_fbench_fftl_target_default "x86-64-v3 (build.sh default)")
  else()
    set(_fbench_fftl_target_default "generic (build.sh default)")
  endif()
  set(FFTN_BENCH_FFTL_CPU_TARGET "${_fbench_fftl_target_default}" CACHE STRING
    "JULIA_CPU_TARGET libfftl was built with (reported by the backend)")
  fbench_add_backend(fftl ${_fbench_fftl_lib})
  target_include_directories(fbench-fftl SYSTEM PRIVATE ${_fbench_fftl_include})
  target_compile_definitions(fbench-fftl PRIVATE
    "FBENCH_FFTL_ROOT=\"${_fbench_fftl_root}\""
    "FBENCH_FFTL_CPU_TARGET=\"${FFTN_BENCH_FFTL_CPU_TARGET}\"")
  list(APPEND _fbench_competitors fftl)
  message(STATUS "fbench: FFTL backend against ${_fbench_fftl_lib}")
else()
  message(STATUS "fbench: FFTL backend not built (set FFTN_BENCH_FFTL_ROOT to a libfftl build)")
endif()

# The harness's reference DFT and check, the driver's statistics, and smoke end to end over
# every backend of this build, under every gate preset (so the sanitizers see the harness).
if(FFTN_BUILD_TESTS)
  add_test(NAME bench.harness_selftest COMMAND fbench-fftn selftest)
  # The FFTN backend's ISA option: scalar, which every build has, is forced and reported,
  # and the cases still pass their check; an unknown name is refused.
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/isa_option_cases.txt
       "id=1 kind=c2c dims=64 batch=3 place=out sign=-1\nid=2 kind=r2c dims=30 batch=2 place=in sign=-1\n")
  add_test(NAME bench.fftn_isa_option
           COMMAND fbench-fftn run --cases ${CMAKE_CURRENT_BINARY_DIR}/isa_option_cases.txt
                   --options ISA=scalar --check-only)
  set_tests_properties(bench.fftn_isa_option PROPERTIES PASS_REGULAR_EXPRESSION
    "\"isa\":\"scalar\",\"isa_forced\":true.*\"id\":\"1\",\"status\":\"ok\",[^\n]*\"pass\":true.*\"id\":\"2\",\"status\":\"ok\",[^\n]*\"pass\":true")
  add_test(NAME bench.fftn_isa_option_refused
           COMMAND fbench-fftn run --cases ${CMAKE_CURRENT_BINARY_DIR}/isa_option_cases.txt
                   --options ISA=sse9 --check-only)
  set_tests_properties(bench.fftn_isa_option_refused PROPERTIES PASS_REGULAR_EXPRESSION
    "\"type\":\"error\",\"message\":\"FFTN options: ISA=")
  if(TARGET fbench-fftw)
    # Includes the check that describe names the real FFTW library, not this executable.
    add_test(NAME bench.harness_selftest_fftw COMMAND fbench-fftw selftest)
  endif()
  find_program(FFTN_UV uv)
  if(FFTN_UV)
    add_test(NAME bench.selftest
             COMMAND ${FFTN_UV} run --frozen --quiet python -m unittest discover -s bench/tests
                     -t bench
             WORKING_DIRECTORY ${PROJECT_SOURCE_DIR})
    add_test(NAME bench.smoke
             COMMAND ${FFTN_UV} run --frozen --quiet python bench/fbench.py
                     --out-dir ${CMAKE_BINARY_DIR}/bench-out smoke --build ${CMAKE_BINARY_DIR}
                     --strict
             WORKING_DIRECTORY ${PROJECT_SOURCE_DIR})
    # The T1 harness end to end in its short mode (every probe reports; no timing claims),
    # and the scheduler's view of the threads catalogue's smoke cases.
    add_test(NAME bench.t1_smoke
             COMMAND ${FFTN_UV} run --frozen --quiet python bench/t1/t1.py
                     --out-dir ${CMAKE_BINARY_DIR}/bench-out run --build ${CMAKE_BINARY_DIR}
                     --rounds 1 --quick --no-quiet-check --strict
             WORKING_DIRECTORY ${PROJECT_SOURCE_DIR})
    add_test(NAME bench.t1_schedule
             COMMAND ${FFTN_UV} run --frozen --quiet python bench/t1/t1.py
                     --out-dir ${CMAKE_BINARY_DIR}/bench-out schedule --build ${CMAKE_BINARY_DIR}
                     --tier smoke -- --threads 4,16 --group 8 --cpus 64
             WORKING_DIRECTORY ${PROJECT_SOURCE_DIR})
    # The OpenMP binaries these run can meet libomp's start-up leak under LeakSanitizer,
    # suppressed as for the OpenMP tests (tests/omp/CMakeLists.txt says why).
    if(_fbench_fftn_omp)
      set_tests_properties(bench.smoke bench.t1_smoke PROPERTIES ENVIRONMENT
        "LSAN_OPTIONS=suppressions=${PROJECT_SOURCE_DIR}/tests/omp/lsan.supp")
    endif()
  else()
    message(STATUS "fbench: uv not found; the driver's tests are not registered")
  endif()
endif()
